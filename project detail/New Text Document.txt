Step 2: Set Up Node.js and Express Backend
Initialize Node.js in your project folder:

npm init -y
Install Required Dependencies:

npm install express mongoose body-parser
Create the app.js File:

This file will handle form submissions and connect to MongoDB.

app.js---javascript

import express from "express";
import mongoose from "mongoose";
import bodyParser from "body-parser";
import path from "path";
import { fileURLToPath } from "url";

// Derive __dirname for ES modules
const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// Initialize Express App
const app = express();
const port = 8000;

// MongoDB Connection
async function connectDB() {
    try {
        await mongoose.connect("mongodb://127.0.0.1:27017/customerData", {
            useNewUrlParser: true,
            useUnifiedTopology: true,
        });
        console.log("Connected to MongoDB successfully!");
    } catch (error) {
        console.error("Error connecting to MongoDB:", error);
    }
}
connectDB();

// Define Mongoose Schema
const contactSchema = new mongoose.Schema({
    name: String,
    email: String,
    findUs: String,
    newsletter: Boolean,
    message: String,
    date: { type: Date, default: Date.now },
});

const Contact = mongoose.model("Contact", contactSchema);

// Middleware
app.use(express.static(__dirname)); // Serve static files like CSS and images
app.use(bodyParser.urlencoded({ extended: true })); // Parse form data

// Routes
// Serve the HTML file
app.get("/", (req, res) => {
    res.sendFile(path.join(__dirname, "index.html"));
});

// Handle form submission
app.post("/contact", async (req, res) => {
    const { name, email, "find-us": findUs, news, message } = req.body;

    const contactData = new Contact({
        name,
        email,
        findUs,
        newsletter: news === "on",
        message,
    });

    try {
        await contactData.save();
        res.send(`
            <h1>Thank you for your submission, ${name}!</h1>
            <a href="/">Go back to the form</a>
        `);
    } catch (error) {
        res.status(500).send("<h1>Something went wrong. Please try again.</h1>");
    }
});

// Start the server
app.listen(port, () => {
    console.log(`Server running at http://localhost:${port}`);
});


// This is change according to you form
